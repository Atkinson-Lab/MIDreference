mt_filtered = mt.filter_rows(
    # Keep rows with call rate ≥ 0.90 and biallelic
    (hl.agg.count_where(hl.is_defined(mt.GT)) / mt.count_cols() >= 0.90) &
    (hl.len(mt.alleles) == 2) &
    (hl.is_snp(mt.alleles[0], mt.alleles[1]))
)

##filter dataset so that non-biallelic variants are fully kicked out
loci = mt_filtered.rows()
loci = loci.group_by(*loci.key).aggregate(n = hl.agg.count())
loci = loci.filter(loci.n > 1)
mt_biallelic_no_ba = mt_filtered.anti_join_rows(loci)
